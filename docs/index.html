2.1 Global walking

Global walking levels differed across countries. Median walking was higher in Mexico and El Salvador and lower in Colombia and Guatemala. Greater variability was observed in Mexico and El Salvador, while extreme values were present in all countries. make_continuous_table <- function(data, variable, label, digits = 1) {
  safe_min <- function(x) {
    if (all(is.na(x))) NA_real_ else min(x, na.rm = TRUE)
  }
  safe_max <- function(x) {
    if (all(is.na(x))) NA_real_ else max(x, na.rm = TRUE)
  }
  total <- data.frame(
    iso2 = "Total",
    median = median(data[[variable]], na.rm = TRUE),
    Q1_val = quantile(data[[variable]], .25, na.rm = TRUE),
    Q3_val = quantile(data[[variable]], .75, na.rm = TRUE),
    min_val = safe_min(data[[variable]]),
    max_val = safe_max(data[[variable]])
  )
  by_iso2 <- data %>%
    filter(!is.na(iso2)) %>%
    group_by(iso2) %>%
    summarise(
      median = median(.data[[variable]], na.rm = TRUE),
      Q1_val = quantile(.data[[variable]], .25, na.rm = TRUE),
      Q3_val = quantile(.data[[variable]], .75, na.rm = TRUE),
      min_val = safe_min(.data[[variable]]),
      max_val = safe_max(.data[[variable]]),
      .groups = "drop"
    )
  stats <- bind_rows(total, by_iso2)
  bind_rows(
    stats %>%
      mutate(
        value = ifelse(
          is.na(median),
          NA_character_,
          paste0(
            format(round(median, digits), trim = TRUE),
            " (",
            format(round(Q1_val, digits), trim = TRUE),
            "; ",
            format(round(Q3_val, digits), trim = TRUE),
            ")"
          )
        ),
        Characteristic = paste0(label, ", median (Q1; Q3)")
      ) %>%
      select(Characteristic, iso2, value),
    stats %>%
      mutate(
        value = ifelse(
          is.na(min_val),
          NA_character_,
          paste0(
            format(round(min_val, digits), trim = TRUE),
            " - ",
            format(round(max_val, digits), trim = TRUE)
          )
        ),
        Characteristic = paste0(label, ", min-max")
      ) %>%
      select(Characteristic, iso2, value)
  )
}
make_categorical_table <- function(data, variable, label) {
  total <- data %>%
    filter(!is.na(.data[[variable]])) %>%
    count(.data[[variable]]) %>%
    mutate(
      iso2 = "Total",
      percent = 100 * n / sum(n),
      value = paste0(
        format(n, big.mark = ","),
        " (",
        formatC(percent, format = "f", digits = 1),
        ")"
      )
    ) %>%
    rename(level = all_of(variable)) %>%
    select(iso2, level, value)
  by_iso2 <- data %>%
    filter(!is.na(.data[[variable]]), !is.na(iso2)) %>%
    group_by(iso2, .data[[variable]]) %>%
    summarise(
      n = n(),
      .groups = "drop"
    ) %>%
    group_by(iso2) %>%
    mutate(
      percent = 100 * n / sum(n),
      value = paste0(
        format(n, big.mark = ","),
        " (",
        formatC(percent, format = "f", digits = 1),
        ")"
      )
    ) %>%
    ungroup() %>%
    rename(level = all_of(variable)) %>%
    select(iso2, level, value)
  bind_rows(total, by_iso2) %>%
    mutate(
      level = as.character(level),
      Characteristic = paste0("   ", label, " - ", level)
    ) %>%
    select(Characteristic, iso2, value) %>%
    bind_rows(
      tibble(
        Characteristic = label,
        iso2 = "Total",
        value = ""
      ),
      .
    )
}

Function 
run_knot_sensitivity <- function(
  outcome_var,
  outcome_label,
  knot_options,
  dta,
  cl
) {
  covars <- c(
    "svyage_scaled",
    "svymale",
    "svyedu",
    "becmedndvinwl2_scaled",
    "becprsbrtl1ad",
    "becpopdensl2_scaled",
    "becadintdensl2_scaled",
    "becparknumpercapl2_scaled",
    "cnssei_l1ad_scaled"
  )
  models <- pblapply(
    seq_along(knot_options),
    function(i) {
      tryCatch({
        valid_idx <- complete.cases(
          dta[, c(
            outcome_var,
            "l1adtemp_pw_1_7_mean",
            "salid1",
            "iso2",
            "quarters",
            "year_c",
            covars
          )]
        )
        knots <- quantile(
          dta$l1adtemp_pw_1_7_mean[valid_idx],
          probs = knot_options[[i]],
          na.rm = TRUE
        )
        formula_model <- as.formula(
          paste0(
            outcome_var,
            " ~ ",
            "splines::ns(l1adtemp_pw_1_7_mean, knots = knots) + ",
            "(1 | salid1) + ",
            "factor(iso2) + ",
            "factor(quarters) + ",
            "year_c + ",
            paste(covars, collapse = " + ")
          )
        )
        glmmTMB(
          formula_model,
          ziformula = ~1,
          data = dta,
          family = nbinom2
        )
      }, error = function(e) {
        message(
          paste(
            "Model error:",
            outcome_label,
            "-",
            names(knot_options)[i],
            "-",
            e$message
          )
        )
        NULL
      })
    },
    cl = cl
  )
  names(models) <- names(knot_options)
  results <- data.frame(
    outcome = outcome_label,
    model = names(models),
    AIC = sapply(
      models,
      function(m) {
        if (is.null(m)) NA else AIC(m)
      }
    )
  ) |>
    dplyr::filter(!is.na(AIC)) |>
    dplyr::arrange(AIC) |>
    dplyr::mutate(
      delta_AIC = AIC - min(AIC)
    )
  return(results)
}
